    if lp.Character then

        end

end)

local function copyOutfitDirectly(targetPlayer)
    local BloxbizRemotes = ReplicatedStorage:FindFirstChild("BloxbizRemotes")
    if not BloxbizRemotes then return false, "لم يتم العثور على BloxbizRemotes" end
    local targetHum = targetPlayer.Character and targetPlayer.Character:FindFirstChildOfClass("Humanoid")
    if not targetHum then return false, "اللاعب غير مترسبن حالياً" end
    local desc = targetHum:GetAppliedDescription()
    if not desc then return false, "فشل جلب بيانات السكن" end

    if BloxbizRemotes:FindFirstChild("CatalogOnResetOutfit") then
        BloxbizRemotes.CatalogOnResetOutfit:FireServer()
        task.wait(0.15)
    end

    if BloxbizRemotes:FindFirstChild("CatalogOnApplyOutfit") then
        local outfitArgs = {
            [1] = {
                ["RightLegColor"] = desc.RightLegColor,
                ["Torso"] = desc.Torso,
                ["ProportionScale"] = desc.ProportionScale,
                ["BodyTypeScale"] = desc.BodyTypeScale,
                ["LeftArmColor"] = desc.LeftArmColor,
                ["Pants"] = desc.Pants,
                ["RightArmColor"] = desc.RightArmColor,
                ["RightArm"] = desc.RightArm,
                ["DepthScale"] = desc.DepthScale,
                ["Head"] = desc.Head,
                ["WidthScale"] = desc.WidthScale,
                ["Shirt"] = desc.Shirt,
                ["LeftLegColor"] = desc.LeftLegColor,
                ["GraphicTShirt"] = desc.GraphicTShirt,
                ["HeadColor"] = desc.HeadColor,
                ["TorsoColor"] = desc.TorsoColor,
                ["RightLeg"] = desc.RightLeg,
                ["HeadScale"] = desc.HeadScale,
                ["HeightScale"] = desc.HeightScale,
                ["LeftArm"] = desc.LeftArm,
                ["LeftLeg"] = desc.LeftLeg,
                ["Accessories"] = desc:GetAccessories(true)
            }
        }
        BloxbizRemotes.CatalogOnApplyOutfit:FireServer(unpack(outfitArgs))
        return true, "تم نسخ سكن @" .. targetPlayer.Name
    end
    return false, "فشل إرسال الأمر للسيرفر"
end

local function StartAttachLoop(mode)
    if TargetPlayerName == "" then
        Notify("خطأ", "يرجى اختيار لاعب من القائمة أولاً!", 3)
        return
    end

    if AttachLoop then AttachLoop:Disconnect(); AttachLoop = nil end
    CurrentMode = mode
    TimeCounter = 0
    
    AttachLoop = RunService.Heartbeat:Connect(function(deltaTime)
        local targetPlayer = Players:FindFirstChild(TargetPlayerName)
        if targetPlayer and targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart") and lp.Character and lp.Character:FindFirstChild("HumanoidRootPart") then
            local targetRoot = targetPlayer.Character.HumanoidRootPart
            local localRoot = lp.Character.HumanoidRootPart
            localRoot.Velocity = Vector3.new(0, 0, 0)
            
            if BangSpeed > 0 then
                TimeCounter = TimeCounter + (deltaTime * BangSpeed * 2)
            else
                TimeCounter = 0
            end
            
            if CurrentMode == "MOUTH" then
                local bangOffset = math.sin(TimeCounter) * 0.75
                localRoot.CFrame = targetRoot.CFrame * CFrame.new(0, 1.25, -0.4 + bangOffset) * CFrame.Angles(0, math.rad(180), 0)
            elseif CurrentMode == "BACK" then
                local bangOffset = math.sin(TimeCounter) * 0.75
                localRoot.CFrame = targetRoot.CFrame * CFrame.new(0, -0.5, 0.4 - bangOffset)
            elseif CurrentMode == "ROTATE" then
                local radius = 2
                local x = math.cos(TimeCounter) * radius
                local z = math.sin(TimeCounter) * radius
                localRoot.CFrame = CFrame.new(targetRoot.Position + Vector3.new(x, 1, z), targetRoot.Position)
            end
        else
            if AttachLoop then AttachLoop:Disconnect(); AttachLoop = nil end
            TargetPlayerName = ""
            Notify("تنبيه", "لقد غادر اللاعب السيرفر أو ماتت شخصيته.", 3)
        end
    end)
end

local http_req = (syn and syn.request) or (http and http.request) or http_request or request

if http_req then
    local currentInputText = ""
    local latestTimestamp = 0
    local firstFetch = true

    local DefaultNameColor = "#CCCCCC"

        ["1The_Invictus"] = {Tag = "Owner", Color = "#FF0000"},
        ["saif_ghzwa"] = {Tag = "Admin", Color = "#00FF00"},

    ChatTab:AddTextbox({
        Name = "Type your message",
        Default = "",
        TextDisappear = true,
        Callback = function(Text) currentInputText = Text end
    })

    local function PlayNotificationSound()
        local sound = Instance.new("Sound")
        sound.SoundId = "rbxassetid://4562203769"
        sound.Volume = 2
        sound.Parent = game:GetService("CoreGui")
        sound:Play()
        game:GetService("Debris"):AddItem(sound, 3)
    end

    local function SendChatMessage()
        if currentInputText == "" or string.gsub(currentInputText, "%s+", "") == "" then return end
        local messageToSend = currentInputText
        currentInputText = ""

        task.spawn(function()
            pcall(function()
                local payload = HttpService:JSONEncode({
                    sender = lp.DisplayName,
                    username = lp.Name,
                    message = messageToSend,
                    timestamp = os.time()
                })

                http_req({
                    Url = DATABASE_URL,
                    Method = "POST",
                    Headers = { ["Content-Type"] = "application/json" },
                    Body = payload
                })
            end)
        end)
    end

    ChatTab:AddButton({
        Name = "📤 Send Message",
        Callback = function() SendChatMessage() end
    })

    local function FetchMessages()
        local fetchUrl = DATABASE_URL .. '?orderBy="$key"&limitToLast=10'
        
        pcall(function()
            local response = http_req({
                Url = fetchUrl,
                Method = "GET"
            })

            if response and response.StatusCode == 200 then
                local data = HttpService:JSONDecode(response.Body)
                if data and type(data) == "table" then
                    local messagesList = {}
                    for key, val in pairs(data) do
                        table.insert(messagesList, val)
                    end
                    
                    table.sort(messagesList, function(a, b)
                        return (a.timestamp or 0) < (b.timestamp or 0)
                    end)

                    local formattedChat = ""
                    local newLatestTimestamp = latestTimestamp

                    for _, msg in ipairs(messagesList) do
                        local msgTime = msg.timestamp or 0
                        local timeStr = os.date("%X", msgTime)
                        local username = msg.username or "Unknown"
                        local displayName = msg.sender or username
                        local textMsg = tostring(msg.message)
                        
                        local tagInfo = CustomTags[username]
                        local nameColor = DefaultNameColor
                        local tagStr = ""
                        
                        if tagInfo then
                            tagStr = "<b>[<font color='" .. tagInfo.Color .. "'>" .. tagInfo.Tag .. "</font>]</b> "
                            nameColor = tagInfo.Color
                        end
                        
                        local formattedSender = tagStr .. "<b><font color='" .. nameColor .. "'>" .. displayName .. "</font></b>"
                        formattedChat = formattedChat .. "[" .. timeStr .. "] " .. formattedSender .. ":\n> " .. textMsg .. "\n\n"

                        if msgTime > newLatestTimestamp then
                            newLatestTimestamp = msgTime
                            if not firstFetch and username ~= lp.Name then
                                PlayNotificationSound()
                                Notify("💬 Message from " .. displayName, textMsg, 4)
                            end
                        end
                    end

                    latestTimestamp = newLatestTimestamp
                    firstFetch = false

                    if formattedChat == "" then formattedChat = "No messages available at the moment." end
                    ChatParagraph:Set("💬 Global Chat (Invictus)", formattedChat)
                end
            end
        end)
    end

    task.spawn(function()
        FetchMessages()
        while task.wait(1) do FetchMessages() end
    end)
else
    ChatTab:AddParagraph("❌ System Error", "Your Executor does not support external HTTP requests, so the chat will not work.")
end

-- Performance & Shaders Engine
local function ClearAllShaders()
    if UltraLoop then UltraLoop:Disconnect(); UltraLoop = nil end
    for _, shader in pairs(Lighting:GetChildren()) do
        if shader:IsA("PostEffect") or shader:IsA("BlurEffect") or shader:IsA("SunRaysEffect") or shader:IsA("ColorCorrectionEffect") or shader:IsA("BloomEffect") or shader:IsA("DepthOfFieldEffect") or shader:IsA("Atmosphere") then
            shader:Destroy()
        end
    end
end

local function NormalAntiLag()
    ClearAllShaders()
    Lighting.GlobalShadows = false
    pcall(function() settings().Rendering.QualityLevel = Enum.QualityLevel.Level05 end)
    for _, obj in pairs(game:GetDescendants()) do
        if obj:IsA("ParticleEmitter") or obj:IsA("Sparkles") then obj.Enabled = false end
    end
end

local function MediumAntiLag()
    ClearAllShaders()
    pcall(function() settings().Rendering.QualityLevel = Enum.QualityLevel.Level03 end)
    Lighting.GlobalShadows = false
    local Terrain = Workspace:FindFirstChildOfClass("Terrain")
    if Terrain then Terrain.WaterWaveSize = 0; Terrain.WaterWaveSpeed = 0 end
    for _, obj in pairs(game:GetDescendants()) do
        if obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Smoke") or obj:IsA("Sparkles") then obj.Enabled = false end
    end
end

local function SuperAntiLag()
    ClearAllShaders()
    pcall(function() settings().Rendering.QualityLevel = Enum.QualityLevel.Level01 end)
    Lighting.GlobalShadows = false
    local Terrain = Workspace:FindFirstChildOfClass("Terrain")
    if Terrain then
        Terrain.WaterWaveSize = 0; Terrain.WaterWaveSpeed = 0
        Terrain.WaterReflectance = 0; Terrain.WaterDetailSize = 0
    end
    for _, obj in pairs(game:GetDescendants()) do
        if obj:IsA("BasePart") or obj:IsA("MeshPart") then
            obj.Material = Enum.Material.SmoothPlastic
            obj.Reflectance = 0; obj.CastShadow = false
        elseif obj:IsA("Texture") or obj:IsA("Decal") then
            obj:Destroy()
        elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Smoke") or obj:IsA("Sparkles") or obj:IsA("Fire") then
            obj.Enabled = false
        end
    end
end

    SuperAntiLag()

                    hum.MoveDirection.X * currentWalkSpeed,
                    root.AssemblyLinearVelocity.Y,
                    hum.MoveDirection.Z * currentWalkSpeed

            else
                root.AssemblyLinearVelocity = Vector3.new(0, root.AssemblyLinearVelocity.Y, 0)

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == UIKeys.Invis then
        IsInvisible = not IsInvisible
        updateInvisParts()
        for _, item in pairs(invisParts) do
            if item.obj and item.obj.Parent then
                item.obj.Transparency = IsInvisible and 0.5 or item.origTrans
            end
        end
    elseif input.KeyCode == UIKeys.Magnet then
        IsHoldingMagnet = true
    end
end)

UserInputService.InputEnded:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == UIKeys.Magnet then
        IsHoldingMagnet = false
    end
end)

-- Initialize Orion Interface
OrionLib:Init()
