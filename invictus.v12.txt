local DATA_FILE = "khurafi_delta_data_v11.json"
local SAVE = {safe1 = "", safe2 = "", safe3 = "", safe4 = "", safe5 = ""}
local SavedWaypoints = {}
local SelectedWaypointName = nil

local function loadData()
   pcall(function()
       if readfile and isfile and isfile(DATA_FILE) then
           local data = HttpService:JSONDecode(readfile(DATA_FILE))
           if data and type(data) == "table" then
               SAVE.safe1 = data.safe1 or ""
               SAVE.safe2 = data.safe2 or ""
               SAVE.safe3 = data.safe3 or ""
               SAVE.safe4 = data.safe4 or ""
               SAVE.safe5 = data.safe5 or ""
               
               if data.waypoints and type(data.waypoints) == "table" then
                   SavedWaypoints = {}
                   for wpName, cfData in pairs(data.waypoints) do
                       if type(cfData) == "table" then
                           SavedWaypoints[wpName] = CFrame.new(unpack(cfData))
                       end
                   end
               end
           end
       end
   end)
end

local function saveData()
   pcall(function()
       if writefile then
           local waypointsData = {}
           for wpName, cf in pairs(SavedWaypoints) do
               waypointsData[wpName] = {cf:GetComponents()}
           end
           local dataToSave = {
               safe1 = SAVE.safe1, safe2 = SAVE.safe2,
               safe3 = SAVE.safe3, safe4 = SAVE.safe4,
               safe5 = SAVE.safe5, waypoints = waypointsData
           }
           writefile(DATA_FILE, HttpService:JSONEncode(dataToSave))
       end
   end)
end
loadData()

local function isSafe(name)
   local lowerName = name:lower()
   local boxes = {SAVE.safe1, SAVE.safe2, SAVE.safe3, SAVE.safe4, SAVE.safe5}
   for _, boxText in ipairs(boxes) do
       if boxText and boxText ~= "" then
           for s in boxText:gmatch("%S+") do
               if lowerName:find(s:lower(), 1, true) then return true end
           end
       end
   end
   return false
end

local function getPlayerNames(includeAll)
   local names = {}
   if includeAll then table.insert(names, "All") end
   for _, p in ipairs(Players:GetPlayers()) do
       if p ~= lp then table.insert(names, p.Name) end
   end
   return names
end

local function getWaypointNames()
   local names = {}
   for name, _ in pairs(SavedWaypoints) do table.insert(names, name) end
   if #names == 0 then table.insert(names, "لا يوجد مواقع محفوظة") end
   return names
end

local HitR = nil
local function findHitRemote()
   local function search(obj)
       for _, child in pairs(obj:GetChildren()) do
           if (child:IsA("RemoteFunction") or child:IsA("RemoteEvent")) and child.Name == "Hit" then return child end
           local found = search(child)
           if found then return found end
       end
       return nil
   end
   HitR = search(ReplicatedStorage)
   if not HitR then
       local combat = ReplicatedStorage:FindFirstChild("Packages")
       if combat then combat = combat:FindFirstChild("Knit") end
       if combat then combat = combat:FindFirstChild("Services") end
       if combat then combat = combat:FindFirstChild("CombatService") end
       if combat then
           local rf = combat:FindFirstChild("RF")
           if rf then HitR = rf:FindFirstChild("Hit") end
       end
   end
end
findHitRemote()

local function doHit(hum, pos)
   if not hum or not hum.Parent then return end
   if HitR then
       if HitR:IsA("RemoteEvent") then HitR:FireServer(hum, Vector3.new(pos.X, pos.Y, pos.Z))
       elseif HitR:IsA("RemoteFunction") then HitR:InvokeServer(hum, Vector3.new(pos.X, pos.Y, pos.Z)) end
   end
end

local function getEspColor(playerName)
   if isSafe(playerName) then return Color3.fromRGB(0, 255, 100) end
   local isAuraTarget = (AuraTargetSelected ~= "All" and AuraTargetSelected ~= "" and playerName == AuraTargetSelected)
   local isGrabTarget = (SpecificTargetEnabled and (playerName == ChosenTargetName1 or playerName == ChosenTargetName2))
   if isAuraTarget or isGrabTarget then return Color3.fromRGB(255, 50, 50) end
   return Color3.fromRGB(255, 255, 255)
end

local function removeEsp(char)
   if not char then return end
   if char:FindFirstChild("ESP_Highlight") then char.ESP_Highlight:Destroy() end
   if char:FindFirstChild("ESP_Billboard") then char.ESP_Billboard:Destroy() end
end

local function applyEsp(player)
   if player == lp or not player.Character then return end
   local char = player.Character
   removeEsp(char)
   if not EspEnabled then return end

   local targetColor = getEspColor(player.Name)

   if EspBoxes and not char:FindFirstChild("ESP_Highlight") then
       local highlight = Instance.new("Highlight")
       highlight.Name = "ESP_Highlight"
       highlight.Adornee = char
       highlight.FillColor = targetColor
       highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
       highlight.FillTransparency = 0.5
       highlight.OutlineTransparency = 0
       highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
       highlight.Parent = char
   end

   local head = char:FindFirstChild("Head") or char:WaitForChild("Head", 3)
   if head and EspNames and not char:FindFirstChild("ESP_Billboard") then
       local billboard = Instance.new("BillboardGui")
       billboard.Name = "ESP_Billboard"
       billboard.Adornee = head
       billboard.Size = UDim2.new(0, 150, 0, 45)
       billboard.StudsOffset = Vector3.new(0, 2.8, 0)
       billboard.AlwaysOnTop = true
       
       local textLabel = Instance.new("TextLabel")
       textLabel.Size = UDim2.new(1, 0, 1, 0)
       textLabel.BackgroundTransparency = 1
       textLabel.TextColor3 = targetColor
       textLabel.TextStrokeTransparency = 0.4
       textLabel.Font = Enum.Font.SourceSansBold
       textLabel.TextSize = 12
       textLabel.Parent = billboard
       billboard.Parent = char

       task.spawn(function()
           while EspEnabled and player and player.Parent and char and char.Parent and head and head.Parent do
               if not char:FindFirstChild("ESP_Billboard") or (EspBoxes and not char:FindFirstChild("ESP_Highlight")) then
                   applyEsp(player)
                   break
               end

               local myChar = lp.Character
               if myChar and myChar:FindFirstChild("HumanoidRootPart") and char:FindFirstChild("HumanoidRootPart") then
                   local dist = math.floor((myChar.HumanoidRootPart.Position - char.HumanoidRootPart.Position).Magnitude)
                   local userName = player.Name
                   local displayName = player.DisplayName
                   
                   if displayName and displayName ~= "" and displayName ~= userName then
                       textLabel.Text = "@" .. userName .. " (" .. displayName .. ")\n[" .. tostring(dist) .. " Studs]"
                   else
                       textLabel.Text = "@" .. userName .. "\n[" .. tostring(dist) .. " Studs]"
                   end
               end
               task.wait(0.4)
           end
       end)
   end
end

local function refreshAllEsp()
   for _, p in ipairs(Players:GetPlayers()) do
       if p ~= lp and p.Character then
           if EspEnabled then applyEsp(p) else removeEsp(p.Character) end
       end
   end
end

local function trackPlayerEsp(plr)
   if plr == lp then return end

   if plr.Character then
       task.spawn(function()
           task.wait(0.5)
           if EspEnabled then applyEsp(plr) end
       end)
   end

   plr.CharacterAdded:Connect(function()
       task.wait(0.5)
       if EspEnabled then applyEsp(plr) end
   end)
end

-- نظام الإشعارات الذكي للـ Safe والـ Target عند الدخول والخروج
local function checkPlayerNotification(plr, eventType)
   local isSafePlayer = isSafe(plr.Name)
   local isTargetPlayer = (AuraTargetSelected ~= "All" and AuraTargetSelected ~= "" and plr.Name == AuraTargetSelected)
                       or (SpecificTargetEnabled and (plr.Name == ChosenTargetName1 or plr.Name == ChosenTargetName2))

   if isSafePlayer or isTargetPlayer then
       local statusTag = isSafePlayer and "🟢 [SAFE]" or "🔴 [TARGET]"
       local actionText = (eventType == "join") and "دخل السيرفر" or "غادر السيرفر"
       local title = statusTag .. " " .. actionText
       local content = "@" .. plr.Name .. " (" .. (plr.DisplayName or "") .. ")"

       pcall(function()
           Rayfield:Notify({
               Title = title,
               Content = content,
               Duration = 4
           })
       end)
   end
end

for _, p in ipairs(Players:GetPlayers()) do
   trackPlayerEsp(p)
end

RunService.Heartbeat:Connect(function(dt)
   if IsInvisible and lp.Character then
       local rootPart = lp.Character:FindFirstChild("HumanoidRootPart")
       local humanoid = lp.Character:FindFirstChild("Humanoid")
       if rootPart and humanoid then
           local cf = rootPart.CFrame
           local camOffset = humanoid.CameraOffset
           local hidden = cf * CFrame.new(0, -200000, 0)
           rootPart.CFrame = hidden
           humanoid.CameraOffset = hidden:ToObjectSpace(CFrame.new(cf.Position)).Position
           RunService.RenderStepped:Wait()
           if rootPart.Parent then 
               rootPart.CFrame = cf
               humanoid.CameraOffset = camOffset
           end
       end
   end

   if isSpeedEnabled and lp.Character then
       local hum = lp.Character:FindFirstChildOfClass("Humanoid")
       if hum and hum.WalkSpeed ~= currentWalkSpeed then hum.WalkSpeed = currentWalkSpeed end
   end

   if AuraEnabled then
       local currentTick = tick()
       if currentTick - lastHitTick >= (1 / apsVal) then
           lastHitTick = currentTick
           local ch = lp.Character
           if ch then
               local myhrp = ch:FindFirstChild("HumanoidRootPart")
               if myhrp then
                   local nearest, nearDist = nil, math.huge
                   for _, p in ipairs(Players:GetPlayers()) do
                       if p ~= lp and p.Character and not isSafe(p.Name) and isTarget(p.Name) then
                           local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                           local hum = p.Character:FindFirstChildOfClass("Humanoid")
                           if hrp and hum and hum.Health > 0 then
                               local dist = (hrp.Position - myhrp.Position).Magnitude
                               if dist <= rangeVal and dist < nearDist then
                                   nearDist = dist; nearest = {hum = hum, pos = hrp.Position + hrp.CFrame.LookVector * 2}
                               end
                           end
                       end
                   end
                   if nearest then task.spawn(function() doHit(nearest.hum, nearest.pos) end) end
               end
           end
       end
   end

   local shouldGrab = AutoGrab
   local activeTargetHRP = nil 

   if AutoTargetEnabled then
       local myChar = lp.Character
       if myChar and myChar:FindFirstChild("HumanoidRootPart") then
           for _, player in ipairs(Players:GetPlayers()) do
               if player ~= lp and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                   if (myChar.HumanoidRootPart.Position - player.Character.HumanoidRootPart.Position).Magnitude <= TriggerDistance then
                       shouldGrab = true
                       activeTargetHRP = player.Character.HumanoidRootPart
                       break
                   end
               end
           end
       end
   end

   if SpecificTargetEnabled then
       local myChar = lp.Character
       if myChar and myChar:FindFirstChild("HumanoidRootPart") then
           local p1 = ChosenTargetName1 ~= "" and Players:FindFirstChild(ChosenTargetName1) or nil
           if p1 and p1.Character and p1.Character:FindFirstChild("HumanoidRootPart") then
               if (myChar.HumanoidRootPart.Position - p1.Character.HumanoidRootPart.Position).Magnitude <= TriggerDistance then
                   shouldGrab = true
                   activeTargetHRP = p1.Character.HumanoidRootPart
               end
           end

           if not activeTargetHRP then
               local p2 = ChosenTargetName2 ~= "" and Players:FindFirstChild(ChosenTargetName2) or nil
               if p2 and p2.Character and p2.Character:FindFirstChild("HumanoidRootPart") then
                   if (myChar.HumanoidRootPart.Position - p2.Character.HumanoidRootPart.Position).Magnitude <= TriggerDistance then
                       shouldGrab = true
                       activeTargetHRP = p2.Character.HumanoidRootPart
                   end
               end
           end
       end
   end

   if lp.Character and lp.Character:FindFirstChild("HumanoidRootPart") then
       local myRoot = lp.Character.HumanoidRootPart
       local currentGrabPower = lp:GetAttribute("GrabPower") or 0
       local isFighting = lp:GetAttribute("Fighting")
       
       if activeTargetHRP then
           if not isFighting then
               if AutoLookEnabled then
                   local targetPos = Vector3.new(activeTargetHRP.Position.X, myRoot.Position.Y, activeTargetHRP.Position.Z)
                   if (targetPos - myRoot.Position).Magnitude > 0.5 then
                       local bg = myRoot:FindFirstChild("AutoLookGyro")
                       if not bg then
                           bg = Instance.new("BodyGyro")
                           bg.Name = "AutoLookGyro"
                           bg.MaxTorque = Vector3.new(0, 400000, 0)
                           bg.P = 30000
                           bg.D = 100
                           bg.Parent = myRoot
                       end
                       bg.CFrame = CFrame.lookAt(myRoot.Position, targetPos)
                   else
                       removeLookGyro()
                   end
               else
                   removeLookGyro()
               end
               
               if AutoMagnetEnabled and currentGrabPower >= 100 then
                   local targetVelocity = activeTargetHRP.AssemblyLinearVelocity
                   local pingPrediction = targetVelocity * 0.05
                   myRoot.CFrame = (activeTargetHRP.CFrame + pingPrediction) * CFrame.new(0, 0, 2.5)
                   
                   if firetouchinterest then
                       firetouchinterest(myRoot, activeTargetHRP, 0)
                       task.wait()
                       firetouchinterest(myRoot, activeTargetHRP, 1)
                   end
               end
           else
               removeLookGyro()
           end

           if IsHoldingMagnet and currentGrabPower >= 100 then
               local targetVelocity = activeTargetHRP.AssemblyLinearVelocity
               local pingPrediction = targetVelocity * 0.05
               myRoot.CFrame = (activeTargetHRP.CFrame + pingPrediction) * CFrame.new(0, 0, 2.5)
               
               if firetouchinterest then
                   firetouchinterest(myRoot, activeTargetHRP, 0)
                   task.wait()
                   firetouchinterest(myRoot, activeTargetHRP, 1)
               end
           end
       else
           removeLookGyro()
       end
   end

   if shouldGrab and GrabRF then
       for i = 1, grabRepeatVal do 
           task.spawn(function() 
               pcall(function() GrabRF:InvokeServer() end) 
           end) 
       end
   end

   if rainbowNameRemote and BWNameActive then
       local now = os.clock()
       rnHue = (rnHue + (dt * rnSpeed)) % 1
       if now - bwLastUpdate >= (currentNameMode == "Police" and 0.12 or rnUpdateRate) then
           local targetColor
           if currentNameMode == "BW" then local intensity = (math.sin(rnHue * math.pi * 2) + 1) / 2; targetColor = Color3.new(intensity, intensity, intensity)
           elseif currentNameMode == "Rainbow" then targetColor = Color3.fromHSV(rnHue, 1, 1)
           elseif currentNameMode == "Pulse" then targetColor = Color3.fromHSV(rnHue, 0.8, 1)
           elseif currentNameMode == "Strobe" then targetColor = Color3.fromRGB(math.random(50,255), math.random(50,255), math.random(50,255))
           elseif currentNameMode == "Police" then policeToggle = not policeToggle; targetColor = policeToggle and Color3.fromRGB(0, 100, 255) or Color3.fromRGB(255, 0, 0) end
           if targetColor then pcall(function() rainbowNameRemote:FireServer(targetColor) end) end
           bwLastUpdate = now
       end
   end
end)

pcall(function()
    Rayfield:LoadConfiguration()
end)
